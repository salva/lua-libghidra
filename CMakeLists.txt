cmake_minimum_required(VERSION 3.26)
project(lua_libghidra_native_readiness LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" -c "import zlib" RESULT_VARIABLE zlib_status)
if(NOT zlib_status EQUAL 0)
  message(FATAL_ERROR "Source build requires Python >=3.10 with zlib (not a runtime dependency)")
endif()
find_package(PkgConfig REQUIRED)
pkg_check_modules(LUA REQUIRED lua5.4)
include(CheckCXXSourceCompiles)
set(CMAKE_REQUIRED_INCLUDES ${LUA_INCLUDE_DIRS})
check_cxx_source_compiles("#include <lua.h>
static_assert(LUA_VERSION_NUM == 504);
static_assert(sizeof(lua_Integer) == 8);
int main() { return 0; }" NATIVE_LUA_ABI_OK)
if(NOT NATIVE_LUA_ABI_OK)
  message(FATAL_ERROR "Requires Lua 5.4 with 64-bit lua_Integer")
endif()
unset(CMAKE_REQUIRED_INCLUDES)
set(NATIVE_INPUT_ROOT "${CMAKE_SOURCE_DIR}/.cache/native-inputs" CACHE PATH "Prepared pinned native inputs")
foreach(pair IN ITEMS "libghidra;libghidra-d51ca813ae1b6b59519551a9be8c0b1ad74d862d" "engine;engine" "protobuf;protobuf" "abseil;abseil" "httplib;httplib")
  list(GET pair 0 name)
  list(GET pair 1 directory)
  string(TOUPPER "${name}" upper)
  set(NATIVE_${upper}_SOURCE "${NATIVE_INPUT_ROOT}/${directory}" CACHE PATH "Override ${name} source")
  if(NOT EXISTS "${NATIVE_${upper}_SOURCE}")
    message(FATAL_ERROR "Missing ${name} input: ${NATIVE_${upper}_SOURCE}; run scripts/prepare-native.py or supply NATIVE_${upper}_SOURCE")
  endif()
endforeach()
set(FETCHCONTENT_FULLY_DISCONNECTED ON CACHE BOOL "No implicit network fetches" FORCE)
set(FETCHCONTENT_SOURCE_DIR_PROTOBUF "${NATIVE_PROTOBUF_SOURCE}" CACHE PATH "" FORCE)
set(FETCHCONTENT_SOURCE_DIR_CPP_HTTPLIB "${NATIVE_HTTPLIB_SOURCE}" CACHE PATH "" FORCE)
set(ABSL_ROOT_DIR "${NATIVE_ABSEIL_SOURCE}" CACHE PATH "" FORCE)
set(GHIDRA_SOURCE_DIR "${NATIVE_ENGINE_SOURCE}" CACHE PATH "" FORCE)
set(LIBGHIDRA_WITH_LOCAL ON CACHE BOOL "" FORCE)
set(LIBGHIDRA_BUILD_PYTHON OFF CACHE BOOL "" FORCE)
set(LIBGHIDRA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
# No ambient static BFD/libiberty archives. Shared Ubuntu binutils closure is
# qualified separately by whole-archive -z defs shared linking below.
find_library(NATIVE_BFD NAMES bfd REQUIRED)
find_library(NATIVE_SFRAME NAMES sframe REQUIRED)
find_library(NATIVE_ZSTD NAMES zstd REQUIRED)
foreach(lib NATIVE_BFD NATIVE_SFRAME NATIVE_ZSTD)
  if(NOT "${${lib}}" MATCHES "\\.so(\\.|$)")
    message(FATAL_ERROR "${lib} must be a qualified shared library, got ${${lib}}")
  endif()
endforeach()
set(LIBGHIDRA_BFD_LIB "${NATIVE_BFD}" CACHE FILEPATH "" FORCE)
set(LIBGHIDRA_SFRAME_LIB "${NATIVE_SFRAME}" CACHE FILEPATH "" FORCE)
set(LIBGHIDRA_ZSTD_LIB "${NATIVE_ZSTD}" CACHE FILEPATH "" FORCE)
# Shared BFD provides its own libiberty implementation; do not silently add .a.
set(LIBGHIDRA_IBERTY_LIB "" CACHE FILEPATH "" FORCE)
add_subdirectory("${NATIVE_LIBGHIDRA_SOURCE}" upstream)
if(NOT LIBGHIDRA_HAS_BFD)
  # Variable is directory scoped upstream: inspect the actual engine target.
  get_target_property(engine_defs ghidra_libdecomp COMPILE_DEFINITIONS)
  if(NOT "LIBGHIDRA_HAS_BFD=1" IN_LIST engine_defs)
    message(FATAL_ERROR "BFD backend is required")
  endif()
endif()
add_library(native_link_closure SHARED tests/native_link_closure.cpp)
target_link_libraries(native_link_closure PRIVATE libghidra::local_whole)
target_link_options(native_link_closure PRIVATE -Wl,-z,defs)
add_executable(native_probe tests/native_probe.cpp)
target_link_libraries(native_probe PRIVATE libghidra::local_whole)
include(CTest)
if(BUILD_TESTING)
  add_executable(native_smoke tests/native_smoke.c)
  target_compile_options(native_smoke PRIVATE -O0 -g -fno-pie)
  target_link_options(native_smoke PRIVATE -no-pie)
  add_test(NAME native_external_constructor COMMAND native_probe "${NATIVE_ENGINE_SOURCE}")
endif()
